  function extractFC2Number(s) {
    const text = String(s || '').trim().toUpperCase();

    const m1 = text.match(/^FC2[-_\s]*PPV[-_\s]*(\d+)$/i);
    if (m1) return m1[1];

    const m2 = text.match(/^FC2[-_\s]*(\d+)$/i);
    if (m2) return m2[1];

    const m3 = text.match(/FC2[-_\s]*PPV[-_\s]*(\d+)/i);
    if (m3) return m3[1];

    const m4 = text.match(/FC2[-_\s]*(\d+)/i);
    if (m4) return m4[1];

    return '';
  }

  function normalizeCode(s) {
    s = String(s || '').trim().toUpperCase().replace(/\s+/g, '');

    if (/^FC2/i.test(s)) {
      const n = extractFC2Number(s);
      if (n) return `FC2-PPV-${n}`;
    }

    const m = s.match(/^([A-Z]{2,12})[-_]?(\d{2,8})$/);
    if (m) return `${m[1]}-${m[2]}`;

    return s;
  }

    const c = normalizeCode(code);

    if (c.startsWith('FC2-PPV-')) {
      const n = extractFC2Number(c);
      return `FC2PPV${n}`;
    }

    return c.replace(/-/g, '');
  }

  function timePrefixToMinute() {
    const d = new Date();
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    const hh = String(d.getHours()).padStart(2, '0');
    const mm = String(d.getMinutes()).padStart(2, '0');

    return `${y}${m}${day}_${hh}${mm}`;
  }

  function uniqueCodesByComparable(codes) {
    const seen = new Set();

      const key = codeComparableKey(code);
      if (seen.has(key)) continue;
      seen.add(key);
      out.push(code);
    }

    return out;
  }

  function parseCSVLine(line) {
    const result = [];
    let cur = '';
    let inQuotes = false;

    for (let i = 0; i < line.length; i++) {
      const ch = line[i];
      const next = line[i + 1];

      if (ch === '"' && inQuotes && next === '"') {
        cur += '"';
        i++;
      } else if (ch === '"') {
        inQuotes = !inQuotes;
      } else if (ch === ',' && !inQuotes) {
        result.push(cur);
        cur = '';
      } else {
        cur += ch;
      }
    }

    result.push(cur);
    return result;
  }

  function mergeCollectionRows(rows) {
    const map = new Map();

    for (const row of rows) {
      const tag = normalizeCollectionTag(row.tag);
      if (!tag || tag === NEED_CHECK_TAG) continue;

      if (!map.has(tag)) {
        map.set(tag, {
          tag,
          codes: []
        });
      }

      const item = map.get(tag);
      item.codes.push(...(row.codes || []));
      item.codes = uniqueCodesByComparable(item.codes);
    }

    return [...map.values()].map(r => ({
      tag: r.tag,
      count: r.codes.length,
      codes: r.codes
    }));
  }

  function parseActressCollectionCSV(text) {
    const lines = text.replace(/^\uFEFF/, '').split(/\r?\n/).filter(x => x.trim());
    if (!lines.length) throw new Error('tag 合集 CSV 是空的');

    const header = parseCSVLine(lines[0]).map(x => x.trim());
    const rows = [];

    const isLegacyFormat =
      header.includes('tag') &&
      header.includes('count') &&
      header.includes('sample_bookmarks');

    if (isLegacyFormat) {
      const tagIdx = header.indexOf('tag');
      const countIdx = header.indexOf('count');
      const sampleIdx = header.indexOf('sample_bookmarks');

      for (const line of lines.slice(1)) {
        const cols = parseCSVLine(line);
        const tag = normalizeCollectionTag(cols[tagIdx] || '');
        if (!tag || tag === NEED_CHECK_TAG) continue;

        const sampleText = cols[sampleIdx] || '';
        const codes = sampleText
          .split(/\s*\|\s*/)
          .map(normalizeCode)
          .filter(Boolean);

        rows.push({
          tag,
          count: Number(cols[countIdx] || 0) || codes.length,
          codes: uniqueCodesByComparable(codes)
        });
      }

      return mergeCollectionRows(rows);
    }

    for (const line of lines.slice(1)) {
      const cols = parseCSVLine(line);
      const tag = normalizeCollectionTag(cols[0] || '');
      if (!tag || tag === NEED_CHECK_TAG) continue;

      const codes = cols
        .slice(2)
        .flatMap(cell => String(cell || '').split(/\s*\|\s*/))
        .map(normalizeCode)
        .filter(Boolean);

      rows.push({
        tag,
        count: Number(cols[1] || 0) || codes.length,
        codes: uniqueCodesByComparable(codes)
      });
    }

    return mergeCollectionRows(rows);
  }

  function buildExistingCodeMap(collectionRows) {
    const map = new Map();

    for (const row of collectionRows) {
      for (const code of row.codes || []) {
        const key = codeComparableKey(code);
        if (!map.has(key)) map.set(key, []);
        map.get(key).push(row.tag);
      }
    }

    return map;
  }

  function normalizeForCompare(s) {
    return cleanTag(s)
      .replace(/[()（）]/g, '')
      .replace(/_/g, '')
      .toLowerCase();
  }

  function pickTagForName(name, collectionRows) {
    const cleaned = cleanRaindropTag(name);
    if (!cleaned) return '';

    const n = normalizeForCompare(cleaned);

    for (const row of collectionRows) {
      const tag = row.tag || '';
      const parts = tag.split('_').map(x => x.trim()).filter(Boolean);

      if (tag === cleaned) return tag;
      if (parts.includes(cleaned)) return tag;

      const tagN = normalizeForCompare(tag);
      if (tagN.includes(n)) return tag;
    }

    return cleaned;
  }

  function mergeNameVariants(names) {
    let arr = [...new Set(names.map(cleanText).filter(x => x && !isBadActressName(x)))];

    arr = arr.filter(name => {
      const n = normalizeForCompare(name);

      return !arr.some(other => {
        if (other === name) return false;
        const o = normalizeForCompare(other);

        return other.length > name.length && o.includes(n);
      });
    });

    return arr;
  }

  function extractActressesFromHTML(pageHtml) {
    const doc = new DOMParser().parseFromString(pageHtml, 'text/html');
    const candidates = [];

    [...doc.querySelectorAll('a[href]')].forEach(a => {
      const href = a.getAttribute('href') || '';
      const text = cleanText(a.textContent || a.innerText || '');

      if (!/\/actresses\//i.test(href)) return;
      if (/\/actors\//i.test(href)) return;
      if (isBadActressName(text)) return;

      candidates.push({
        raw: text,
        tagName: cleanRaindropTag(text),
        href
      });
    });

    const seen = new Set();

    return candidates.filter(x => {
      const key = `${x.tagName}|${x.href}`;
      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    });
  }

  function extractTypeTagsFromHTML(pageHtml) {
    const doc = new DOMParser().parseFromString(pageHtml, 'text/html');
    const result = [];

    function addTypeTag(text) {
      const raw = cleanText(text);
      if (!raw || isBadTypeTag(raw)) return;

      const tag = cleanRaindropTag(raw);
      if (tag && !result.includes(tag)) {
        result.push(tag);
      }
    }

    function isTypeLabelText(text) {
      const s = cleanText(text).replace(/\s/g, '');
      return /^(类型|類型|类别|類別)[:：]?$/.test(s);
    }

    function collectGenreAnchorsFrom(container) {
      if (!container) return [];

      return [...container.querySelectorAll('a[href]')].filter(a => {
        const href = a.getAttribute('href') || '';
        const text = cleanText(a.textContent || a.innerText || '');

        if (!text || isBadTypeTag(text)) return false;

        return /\/genres?\//i.test(href);
      });
    }

    const allNodes = [...doc.querySelectorAll('div, span, p, li, dt, dd')];

    for (const node of allNodes) {
      const ownText = cleanText(node.childNodes.length === 1 ? node.textContent : '');

      if (!isTypeLabelText(ownText)) continue;

      let cur = node;

      for (let depth = 0; depth < 5 && cur; depth++) {
        const anchors = collectGenreAnchorsFrom(cur.parentElement);

        for (const a of anchors) {
          addTypeTag(a.textContent || '');
        }

        if (result.length) break;
        cur = cur.parentElement;
      }

      if (result.length) break;
    }

    if (!result.length) {
      const blocks = [...doc.querySelectorAll('div, p, li, section')];

      for (const block of blocks) {
        const text = cleanText(block.textContent || '');

        if (!text || text.length > 180) continue;
        if (!/(类型|類型|类别|類別)\s*[：:]/i.test(text)) continue;

        const anchors = collectGenreAnchorsFrom(block);

        for (const a of anchors) {
          addTypeTag(a.textContent || '');
        }

        if (result.length) break;
      }
    }

    return [...new Set(result)];
  }

  function pageLooksPlayable(pageHtml) {
    return /<video|m3u8|plyr|player|iframe|播放|play/i.test(pageHtml);
  }

  function pageContainsCode(pageHtml, code) {
    const compact = pageHtml.toUpperCase().replace(/[_\s]/g, '-');
    const compactLoose = compact.replace(/-/g, '');

    return codeVariants(code).some(v => {
      const vv = v.toUpperCase();
      return compact.includes(vv) || compactLoose.includes(vv.replace(/-/g, ''));
    });
  }

  async function fetchText(url) {
    let lastErr;

    for (let i = 0; i <= MAX_RETRY; i++) {
      try {
        const res = await fetch(url, { credentials: 'include' });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return await res.text();
      } catch (e) {
        lastErr = e;
        await sleep(1000);
      }
    }

    throw lastErr;
  }

  async function findWorkingPage(code) {
    const urls = candidateUrls(code);

    for (const url of urls) {
      try {
        const html = await fetchText(url);
        const hasCode = pageContainsCode(html, code);
        const playable = pageLooksPlayable(html);
        const actresses = extractActressesFromHTML(html);
        const typeTags = extractTypeTagsFromHTML(html);

        if (hasCode && actresses.length) {
          return {
            url,
            status: playable ? 'ok' : 'page_ok_play_unknown',
            notes: playable ? 'page_and_player_structure_found' : 'page_found_but_player_uncertain',
            actresses,
            typeTags
          };
        }

        if (hasCode) {
          return {
            url,
            status: playable ? 'no_actress_found' : 'need_manual_check',
            notes: playable ? 'page_found_but_no_actress_field' : 'page_found_but_no_actress_and_player_uncertain',
            actresses,
            typeTags
          };
        }
      } catch (e) {}

      await sleep(250);
    }

    return {
      url: candidateUrls(code)[0],
      status: 'not_found',
      notes: 'all_candidate_urls_failed_or_not_matching_code',
      actresses: [],
      typeTags: []
    };
  }

  function addProblemTags(row) {
    const problemStatuses = new Set([
      'not_found',
      'need_manual_check'
    ]);

    const tags = row.tags
      ? row.tags.split(',').map(x => x.trim()).filter(Boolean)
      : [];

    const actressTags = row.actress_tags
      ? row.actress_tags.split(',').map(x => x.trim()).filter(Boolean)
      : [];

    const hasRealActress = actressTags.some(t => t && t !== UNKNOWN_ACTRESS_TAG);

    const isProblem = problemStatuses.has(row.status);
    const isUnknownActress = !hasRealActress;

    if (isUnknownActress && !tags.includes(UNKNOWN_ACTRESS_TAG)) {
      tags.unshift(UNKNOWN_ACTRESS_TAG);
    }

    if (isProblem && !tags.includes(NEED_CHECK_TAG)) {
      tags.push(NEED_CHECK_TAG);
    }

    const finalActressTags = isUnknownActress
      ? [UNKNOWN_ACTRESS_TAG]
      : actressTags;

    const classifiedRow = {
      ...row,
      tags: [...new Set(tags)].join(','),
      actress_tags: finalActressTags.join(','),
      include_in_import: true
    };
    const referenceMatches = referenceMatchesForRow(classifiedRow);
    const exportBlacklistMatches = exportBlacklistMatchesForRow(classifiedRow);
    const importableRow = {
      ...classifiedRow,
      include_in_import: exportBlacklistMatches.length === 0
    };
    const targetFolder = targetFolderForRow(importableRow);

    const extraNotes = [];

    if (isProblem) extraNotes.push(`added_${NEED_CHECK_TAG}_tag`);
    if (isUnknownActress) extraNotes.push(`added_${UNKNOWN_ACTRESS_TAG}_tag`);
    if (exportBlacklistMatches.length) extraNotes.push('excluded_by_raindrop_export_blacklist');
    else extraNotes.push(`placed_in_${targetFolder}`);

    return {
      ...importableRow,
      matched_tag: finalActressTags.join(','),
      needs_lookup: needsLookup(classifiedRow),
      reference_matches: referenceMatches.join(','),
      export_blacklist_matches: exportBlacklistMatches.join(','),
      excluded_from_raindrop: exportBlacklistMatches.length > 0,
      target_folder: targetFolder,
      notes: extraNotes.length
        ? `${row.notes}; ${extraNotes.join('; ')}`
        : row.notes
    };
  }

  function updateActressCollectionWithRows(collectionRows, processedRows) {
    const rows = collectionRows.map(r => ({
      tag: normalizeCollectionTag(r.tag),
      codes: uniqueCodesByComparable(r.codes || [])
    })).filter(r => r.tag && r.tag !== NEED_CHECK_TAG);

    const index = new Map(rows.map((r, i) => [r.tag, i]));

    const addedNewActressRows = [];
    const updatedActressRows = [];

    for (const row of processedRows) {
      if (!row.include_in_import) continue;

      const code = normalizeCode(row.code);
      const actressTags = row.actress_tags
        ? row.actress_tags.split(',').map(x => normalizeCollectionTag(x.trim())).filter(Boolean)
        : [];

      for (const tag of actressTags) {
        if (!tag || tag === NEED_CHECK_TAG) continue;

        if (!index.has(tag)) {
          rows.push({
            tag,
            codes: [code]
          });
          index.set(tag, rows.length - 1);
          addedNewActressRows.push(tag);
        } else {
          const item = rows[index.get(tag)];
          const keys = item.codes.map(codeComparableKey);

          if (!keys.includes(codeComparableKey(code))) {
            item.codes.push(code);
            updatedActressRows.push(tag);
          }
        }
      }
    }

    return {
      rows: mergeCollectionRows(rows).map(r => {
        const codes = uniqueCodesByComparable(r.codes || []);

        return {
          tag: r.tag,
          count: codes.length,
          codes
        };
      }),
      addedNewActressRows: [...new Set(addedNewActressRows)],
      updatedActressRows: [...new Set(updatedActressRows)]
    };
  }

  async function exportFiles({
    finalRows,
    reportRows,
    updatedCollectionRows,
    outputDirInfo,
    addedNewActressRows,
    updatedActressRows,
    duplicateInputRows,
    alreadyExistsRows

    const csvFileName = `${prefix}_missav_raindrop_import_threeway.csv`;
    const reportFileName = `${prefix}_missav_import_report_threeway.csv`;
    const collectionFileName = `${prefix}_女优tag合集.csv`;
    const backupFileName = `${prefix}_missav_actress_fetch_backup_threeway.json`;

    const importRows = finalRows.filter(r => r.include_in_import);
    const referenceRows = importRows.filter(r => targetFolderForRow(r) === REFERENCE_HIT_FOLDER_NAME);
    const needCheckRows = importRows.filter(r => targetFolderForRow(r) === NEED_CHECK_FOLDER_NAME);
    const otherRows = importRows.filter(r => targetFolderForRow(r) === OTHER_FOLDER_NAME);

    function bookmarkLines(rows) {
      return rows.map(r =>
        `    <DT><A HREF="${escapeHtml(r.url)}" TAGS="${escapeHtml(r.tags)}">${escapeHtml(r.code)}</A>`
      ).join('\n');
    }

    function folderBlock(name, rows) {
      return `
  <DT><H3>${escapeHtml(name)}</H3>
  <DL><p>
${bookmarkLines(rows)}
  </DL><p>
`;
    }

    const htmlText = `<!DOCTYPE NETSCAPE-Bookmark-file-1>
<META HTTP-EQUIV="Content-Type" CONTENT="text/html; charset=UTF-8">
<TITLE>MissAV Three-Way Import</TITLE>
<H1>MissAV Three-Way Import</H1>

<DL><p>
${folderBlock(REFERENCE_HIT_FOLDER_NAME, referenceRows)}
${folderBlock(NEED_CHECK_FOLDER_NAME, needCheckRows)}
${folderBlock(OTHER_FOLDER_NAME, otherRows)}
</DL><p>
`;

    const csvText = [
      [
        'url',
        'title',
        'tags',
        'actress_tags',
        'type_tags',
        'status',
        'needs_lookup',
        'reference_matches',
        'excluded_from_raindrop',
        'export_blacklist_matches',
        'target_folder',
        'actress_raw',
        'matched_tag',
        'notes'
      ].join(','),
      ...importRows.map(r => [
        csvCell(r.url),
        csvCell(r.code),
        csvCell(r.tags),
        csvCell(r.actress_tags),
        csvCell(r.type_tags),
        csvCell(r.status),
        csvCell(needsLookup(r) ? 'yes' : 'no'),
        csvCell(referenceMatchesForRow(r).join(',')),
        csvCell(exportBlacklistMatchesForRow(r).length ? 'yes' : 'no'),
        csvCell(exportBlacklistMatchesForRow(r).join(',')),
        csvCell(targetFolderForRow(r)),
        csvCell(r.actress_raw),
        csvCell(r.matched_tag),
        csvCell(r.notes)
      ].join(','))
    ].join('\n');

    const reportText = [
      [
        'code',
        'input_code',
        'url',
        'status',
        'include_in_import',
        'needs_lookup',
        'reference_matches',
        'excluded_from_raindrop',
        'export_blacklist_matches',
        'target_folder',
        'issue',
        'existing_tags',
        'actress_raw',
        'actress_tags',
        'type_tags',
        'final_tags'
      ].join(','),
      ...reportRows.map(r => {
        return [
          csvCell(r.code),
          csvCell(r.input_code || r.code),
          csvCell(r.url || ''),
          csvCell(r.status),
          csvCell(r.include_in_import ? 'yes' : 'no'),
          csvCell(r.include_in_import && needsLookup(r) ? 'yes' : 'no'),
          csvCell(r.include_in_import ? referenceMatchesForRow(r).join(',') : ''),
          csvCell(exportBlacklistMatchesForRow(r).length ? 'yes' : 'no'),
          csvCell(exportBlacklistMatchesForRow(r).join(',')),
          csvCell(r.include_in_import ? targetFolderForRow(r) : ''),
          csvCell(r.notes || ''),
          csvCell(r.existing_tags || ''),
          csvCell(r.actress_raw || ''),
          csvCell(r.actress_tags || ''),
          csvCell(r.type_tags || ''),
          csvCell(r.tags || '')
        ].join(',');
      })
    ].join('\n');

    const sortedCollectionRows = sortCollectionRowsForExport(updatedCollectionRows);

    const maxCodes = Math.max(0, ...sortedCollectionRows.map(r => r.codes.length));
    const collectionHeader = [
      '女优tag名字',
      '收藏数',
      ...Array.from({ length: maxCodes }, (_, i) => `番号${i + 1}`)
    ];

    const collectionText = [
      collectionHeader.map(csvCell).join(','),
      ...sortedCollectionRows.map(r => [
        csvCell(r.tag),
        csvCell(r.count),
        ...r.codes.map(csvCell)
      ].join(','))
    ].join('\n');

    const backup = {
      finalRows,
      reportRows,
      updatedCollectionRows: sortedCollectionRows,
      addedNewActressRows,
      updatedActressRows,
      duplicateInputRows,
      alreadyExistsRows,
      referenceHitImportCount: referenceRows.length,
      needCheckImportCount: needCheckRows.length,
      otherImportCount: otherRows.length,
      referenceActressTagCount: REFERENCE_ACTRESS_TAG_SET.size,
      raindropExportBlacklistTagCount: RAINDROP_EXPORT_BLACKLIST_TAG_SET.size,
      excludedFromRaindropCount: finalRows.filter(r => exportBlacklistMatchesForRow(r).length).length,
      needCheckTag: NEED_CHECK_TAG,
      unknownActressTag: UNKNOWN_ACTRESS_TAG,
      referenceHitFolderName: REFERENCE_HIT_FOLDER_NAME,
      needCheckFolderName: NEED_CHECK_FOLDER_NAME,
      otherFolderName: OTHER_FOLDER_NAME,
      updatedCollectionFileName: collectionFileName,
      outputFolderName: outputDirInfo?.folderName || '',
      exportedAt: new Date().toISOString()
    };

    await saveTextFile(htmlFileName, htmlText, 'text/html;charset=utf-8', outputDirInfo);
    await saveTextFile(csvFileName, '\ufeff' + csvText, 'text/csv;charset=utf-8', outputDirInfo);
    await saveTextFile(reportFileName, '\ufeff' + reportText, 'text/csv;charset=utf-8', outputDirInfo);
    await saveTextFile(collectionFileName, '\ufeff' + collectionText, 'text/csv;charset=utf-8', outputDirInfo);
    await saveTextFile(backupFileName, JSON.stringify(backup, null, 2), 'application/json;charset=utf-8', outputDirInfo);

    console.log('Raindrop 导入文件：', htmlFileName);
    console.log('本次处理报告：', reportFileName);
    console.log('更新后的完整女优 tag 合集：', collectionFileName);
    console.log(`${REFERENCE_HIT_FOLDER_NAME} 数量：`, referenceRows.length);
    console.log(`${NEED_CHECK_FOLDER_NAME} 数量：`, needCheckRows.length);
    console.log(`${OTHER_FOLDER_NAME} 数量：`, otherRows.length);
  }

  const runState = await prepareRunByUserClick();

  const oldCollectionFile = runState.oldCollectionFile;
  const outputDirInfo = runState.outputDirInfo;

  const collectionRows = parseActressCollectionCSV(oldCollectionFile.text);
  const existingCodeMap = buildExistingCodeMap(collectionRows);

  console.log('已选择旧 tag 合集：', oldCollectionFile.name);
  console.log('旧 tag 合集女优行数：', collectionRows.length);
  console.log('旧 tag 合集已收录唯一番号数：', existingCodeMap.size);

  const rawInputCodes = CODE_TEXT
    .split(/\r?\n/)
    .map(x => x.trim())
    .filter(Boolean);

  const inputSeen = new Map();
  const codesToProcess = [];
  const reportRows = [];
  const finalRows = [];
  const duplicateInputRows = [];
  const alreadyExistsRows = [];

  for (const inputCode of rawInputCodes) {
    const code = normalizeCode(inputCode);
    const key = codeComparableKey(code);

    if (inputSeen.has(key)) {
      const row = {
        input_code: inputCode,
        code,
        url: '',
        status: 'duplicate_in_input',
        include_in_import: false,
        notes: `本次输入重复，第一次出现为：${inputSeen.get(key)}`,
        existing_tags: '',
        actress_raw: '',
        actress_tags: '',
        type_tags: '',
        tags: '重复输入'
      };

      duplicateInputRows.push(row);
      reportRows.push(row);

      if (SKIP_DUPLICATE_IN_INPUT) continue;
    }

    inputSeen.set(key, code);

    if (existingCodeMap.has(key)) {
      const existingTags = [...new Set(existingCodeMap.get(key))].join(',');

      const row = {
        input_code: inputCode,
        code,
        url: candidateUrls(code)[0],
        status: 'already_exists_in_tag_collection',
        include_in_import: false,
        notes: '已存在于旧 tag 合集，不重复导入',
        existing_tags: existingTags,
        actress_raw: '',
        actress_tags: existingTags,
        type_tags: '',
        tags: '已存在'
      };

      alreadyExistsRows.push(row);
      reportRows.push(row);

      if (SKIP_ALREADY_EXISTS) continue;
    }

    codesToProcess.push({
      input_code: inputCode,

    });
  }

  console.log('本次输入番号数：', rawInputCodes.length);
  console.log('本次输入内部重复数：', duplicateInputRows.length);
  console.log('旧 tag 合集中已存在数：', alreadyExistsRows.length);
  console.log('本次需要实际抓取数：', codesToProcess.length);

      break;
    }

    const item = codesToProcess[i];
    const code = item.code;

    console.log(`[${i + 1}/${codesToProcess.length}] 正在处理：${code}`);

    const result = await findWorkingPage(code);

    const names = mergeNameVariants(result.actresses.map(x => x.raw));
    let actressTags = [...new Set(names.map(name => pickTagForName(name, collectionRows)).filter(Boolean))];

    })));
  }

  if (needCheckRows.length) {
    console.log(`进入「${NEED_CHECK_FOLDER_NAME}」文件夹的番号：`);
    console.table(needCheckRows.map(r => ({
      code: r.code,
      status: r.status,
      tags: r.tags,
      issue: r.notes
    })));
  }

  if (updateResult.addedNewActressRows.length) {
    console.log('本次新增女优行：', updateResult.addedNewActressRows);
  }

  if (updateResult.updatedActressRows.length) {
    console.log('本次增量更新已有女优行：', updateResult.updatedActressRows);
  }
})();
